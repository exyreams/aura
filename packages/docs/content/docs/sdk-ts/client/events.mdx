AURA emits four typed Anchor events from its instruction handlers. The SDK
exports typed event definitions, discriminator constants, and a helper for
matching raw log bytes.

```typescript
import {
  AuraEvents,
  TreasuryAuditEvent,
  ProposalLifecycleEvent,
  ExecutionLifecycleEvent,
  EventDiscriminator,
  matchesEventDiscriminator,
} from '@aura-protocol/sdk-ts';
```

## Event Types

### TreasuryAuditEvent

Emitted once after every instruction that mutates treasury state. Clients and
indexers subscribe to this event to build an off-chain audit trail.

```typescript
type TreasuryAuditEvent = {
  treasury: PublicKey;
  kind: string;      // instruction label, e.g. "create_treasury", "propose_transaction"
  detail: string;    // human-readable summary of the change
  timestamp: BN;
};
```

### ProposalLifecycleEvent

Emitted after every proposal state change: propose, execute, deny, cancel,
and expire. Allows clients to track proposal status without polling the account.

```typescript
type ProposalLifecycleEvent = {
  treasury: PublicKey;
  proposalId: BN;
  proposalDigest: string;
  status: number;   // 0=Pending, 1=Approved, 2=Denied, 3=Cancelled, 4=Expired
  approved: boolean;
  violation: number; // violation code if denied
};
```

### ExecutionLifecycleEvent

Emitted after `finalize_execution` completes. Carries the full execution
outcome including signature and decryption account references for off-chain
verification.

```typescript
type ExecutionLifecycleEvent = {
  treasury: PublicKey;
  proposalId: BN;
  proposalDigest: string;
  finalStatus: number;
  approved: boolean;
  violation: number;
  messageApprovalId: string | null;
  messageApprovalAccount: string | null;
  decryptionRequestId: string | null;
  decryptionRequestAccount: string | null;
};
```

### PolicyReceiptWrittenEvent

Emitted when `write_policy_receipt` stores an immutable policy snapshot for a
proposal.

```typescript
type PolicyReceiptWrittenEvent = {
  treasury: PublicKey;
  proposalId: BN;
  policyVersion: number;
  decision: number;
  primaryViolation: number;
  requiredApprovalLevel: number;
  policyAttested: boolean;
};
```

## Event Discriminators

Eight-byte prefixes used to identify event type in raw transaction logs.

```typescript
import { EventDiscriminator } from '@aura-protocol/sdk-ts';

// Buffers: [209, 27, 57, 147, 169, 125, 166, 58]
EventDiscriminator.treasuryAuditEvent

// Buffer: [198, 23, 28, 210, 232, 47, 7, 199]
EventDiscriminator.proposalLifecycleEvent

// Buffer: [170, 155, 187, 106, 242, 102, 71, 103]
EventDiscriminator.executionLifecycleEvent
```

### `matchesEventDiscriminator`

Check whether raw event data starts with a given discriminator:

```typescript
function matchesEventDiscriminator(data: Buffer, discriminator: Buffer): boolean
```

```typescript
import { EventDiscriminator, matchesEventDiscriminator } from '@aura-protocol/sdk-ts';

const data = Buffer.from(rawEventBytes);
if (matchesEventDiscriminator(data, EventDiscriminator.proposalLifecycleEvent)) {
  // decode as ProposalLifecycleEvent
}
```

## Parsing Events from a Transaction

Use `AuraClient.coder.events.decode()` to parse events from a confirmed
transaction's logs:

```typescript
import { AuraClient, AURA_PROGRAM_ID } from '@aura-protocol/sdk-ts';
import { Connection } from '@solana/web3.js';

const client = new AuraClient({ connection });

async function parseTransactionEvents(signature: string) {
  const tx = await connection.getTransaction(signature, {
    commitment: 'confirmed',
    maxSupportedTransactionVersion: 0,
  });

  if (!tx?.meta?.logMessages) return [];

  return tx.meta.logMessages
    .map((log) => {
      try {
        return client.coder.events.decode(log);
      } catch {
        return null;
      }
    })
    .filter(Boolean);
}

const events = await parseTransactionEvents(txSignature);
for (const event of events) {
  if (!event) continue;

  switch (event.name) {
    case 'proposalLifecycleEvent': {
      const data = event.data as ProposalLifecycleEvent;
      console.log('Proposal:', data.proposalId.toString());
      console.log('Approved:', data.approved, 'Status:', data.status);
      break;
    }
    case 'executionLifecycleEvent': {
      const data = event.data as ExecutionLifecycleEvent;
      console.log('Finalized proposal:', data.proposalId.toString());
      break;
    }
    case 'treasuryAuditEvent': {
      const data = event.data as TreasuryAuditEvent;
      console.log(`Audit [${data.kind}]:`, data.detail);
      break;
    }
  }
}
```

## Subscribing to Program Logs (WebSocket)

Subscribe to all AURA events in real time using the Solana WebSocket API:

```typescript
import {
  AuraClient,
  AURA_PROGRAM_ID,
  type ProposalLifecycleEvent,
  type TreasuryAuditEvent,
} from '@aura-protocol/sdk-ts';
import { Connection } from '@solana/web3.js';

const wsConnection = new Connection(
  'wss://devnet.helius-rpc.com/?api-key=YOUR_KEY',
  'confirmed',
);
const client = new AuraClient({ connection: wsConnection });

const subscriptionId = wsConnection.onLogs(
  AURA_PROGRAM_ID,
  ({ logs, signature, err }) => {
    if (err) return;

    for (const log of logs) {
      try {
        const event = client.coder.events.decode(log);
        if (!event) continue;

        switch (event.name) {
          case 'treasuryAuditEvent': {
            const data = event.data as TreasuryAuditEvent;
            console.log(`[${data.kind}] ${data.treasury.toBase58()} — ${data.detail}`);
            break;
          }
          case 'proposalLifecycleEvent': {
            const data = event.data as ProposalLifecycleEvent;
            console.log(`Proposal ${data.proposalId}: approved=${data.approved}`);
            break;
          }
          case 'executionLifecycleEvent': {
            const data = event.data as ExecutionLifecycleEvent;
            console.log(`Execution finalized: proposal ${data.proposalId}`);
            break;
          }
        }
      } catch {
        // non-AURA log line, ignore
      }
    }
  },
  'confirmed',
);

// Later: unsubscribe
await wsConnection.removeOnLogsListener(subscriptionId);
```
